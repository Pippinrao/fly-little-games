cmake_minimum_required(VERSION 3.22)
project(nearby_probe LANGUAGES CXX)
set(PROBE_ARCHIVE "${CMAKE_CURRENT_LIST_DIR}/deps/libnearby_quic_spike.a")
if(NOT EXISTS "${PROBE_ARCHIVE}" OR NOT EXISTS "${CMAKE_CURRENT_LIST_DIR}/deps/nearby_quic_spike.h")
  message(FATAL_ERROR "Stage the actual OHOS Rust archive/header with stage-harmony.ps1 first")
endif()
add_library(nearby_quic_rust STATIC IMPORTED)
set_target_properties(nearby_quic_rust PROPERTIES IMPORTED_LOCATION "${PROBE_ARCHIVE}")
add_library(nearbyprobe SHARED napi_init.cpp)
target_compile_features(nearbyprobe PRIVATE cxx_std_17)
target_include_directories(nearbyprobe PRIVATE "${CMAKE_CURRENT_LIST_DIR}/deps")
set_target_properties(nearbyprobe PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_compile_options(nearbyprobe PRIVATE -Wall -Wextra -Werror)
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS" AND CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  target_compile_options(nearbyprobe PRIVATE -Wno-error=unused-command-line-argument)
endif()
target_link_libraries(nearbyprobe PRIVATE nearby_quic_rust libace_napi.z.so)
target_link_options(nearbyprobe PRIVATE -Wl,--no-undefined -Wl,--exclude-libs,ALL)
